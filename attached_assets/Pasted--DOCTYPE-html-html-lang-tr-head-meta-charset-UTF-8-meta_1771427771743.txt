let basket = { items: {}, payId: 'nakit' };

function render() {
    document.getElementById('main-product-area').innerHTML = `<div class="section-title">Ana Ürün</div><div class="ios-list">${DATA.main.map(p => `<div class="list-item"><div><b>${p.name}</b><br><small style="color:var(--ios-blue)">${p.price} TL</small></div><div class="qty-box"><button onclick="updateQty('${p.id}', -1, true)">-</button><div class="val">${basket.items[p.id] || 0}</div><button onclick="updateQty('${p.id}', 1, true)">+</button></div></div>`).join('')}</div>`;

    document.getElementById('categories-area').innerHTML = DATA.categories.map(cat => `<div class="section-title">${cat.title}</div><div class="product-grid">${cat.items.map(item => `<div class="grid-item ${basket.items[item.id] > 0 ? 'active' : ''}"><div class="name">${item.name}</div><img class="product-img" src="${item.img.trim()}"><div class="price">${item.price} TL</div><div class="qty-box"><button onclick="updateQty('${item.id}', -1, false)">-</button><div class="val">${basket.items[item.id] || 0}</div><button onclick="updateQty('${item.id}', 1, false)">+</button></div></div>`).join('')}</div>`).join('');

    calculate();
}

function updateQty(id, val, isMain) {
    let next = (basket.items[id] || 0) + val;
    if (next <= 0) delete basket.items[id]; else basket.items[id] = next;
    
    render();

    // Sadece Ana Ürün miktar artırıldığında kaydır
    if (isMain && val > 0) {
        setTimeout(() => {
            document.getElementById('payment-anchor').scrollIntoView({ 
                behavior: 'smooth', 
                block: 'start' 
            });
        }, 150);
    }
}

function calculate() {
    let subtotal = 0;
    const all = [...DATA.main, ...DATA.categories.flatMap(c => c.items)];
    const selected = all.filter(p => basket.items[p.id] > 0);
    selected.forEach(p => subtotal += basket.items[p.id] * p.price);

    const minPerc = Math.min((subtotal / CONFIG.minLimit) * 100, 100);
    const shipPerc = Math.min((subtotal / CONFIG.shipLimit) * 100, 100);
    
    document.getElementById('top-min-bar').style.width = minPerc + "%";
    document.getElementById('top-ship-bar').style.width = shipPerc + "%";
    document.getElementById('top-min-text').innerText = `${subtotal}/${CONFIG.minLimit} TL`;
    document.getElementById('top-ship-text').innerText = `${subtotal}/${CONFIG.shipLimit} TL`;

    const minHint = document.getElementById('min-needed');
    const shipHint = document.getElementById('ship-needed');

    if (subtotal > 0) {
        const mDiff = CONFIG.minLimit - subtotal;
        const sDiff = CONFIG.shipLimit - subtotal;
        minHint.innerText = mDiff > 0 ? `Minimum sipariş için ${mDiff} TL kaldı.` : "✅ Minimum tutar aşıldı.";
        shipHint.innerText = sDiff > 0 ? `Bedava kargo için ${sDiff} TL kaldı.` : "🚚 Kargo artık bedava!";
    } else {
        minHint.innerText = ""; shipHint.innerText = "";
    }

    const currentShipFee = (subtotal > 0 && subtotal < CONFIG.shipLimit) ? CONFIG.shipFee : 0;
    document.getElementById('shipping-display').innerText = currentShipFee === 0 ? (subtotal === 0 ? "0 TL" : "ÜCRETSİZ") : currentShipFee + " TL";

    document.getElementById('basket-items-list').innerHTML = selected.length > 0 ? selected.map(p => `<div style="display:flex; justify-content:space-between; font-size:12px; margin-bottom:4px;"><span>${p.name} (x${basket.items[p.id]})</span><span>${(p.price * basket.items[p.id])} TL</span></div>`).join('') : "<div style='text-align:center; color:gray; font-size:12px;'>Sepet henüz boş</div>";

    document.getElementById('payment-list').innerHTML = DATA.payments.map(p => {
        let totalWithShip = Math.round(subtotal * (1 - p.disc)) + currentShipFee;
        return `<div class="list-item" onclick="basket.payId='${p.id}'; render();" style="cursor:pointer">
            <div style="display:flex; align-items:center; gap:10px;">
                <div style="width:18px; height:18px; border-radius:50%; border:1px solid #ccc; background:${basket.payId === p.id ? 'var(--ios-blue)' : 'white'}"></div>
                <span>${p.name} ${p.disc > 0 ? `<small style="color:var(--ios-green)">${p.tag}</small>` : ''}</span>
            </div>
            <div style="font-weight:700;">${subtotal > 0 ? totalWithShip : 0} TL</div>
        </div>`;
    }).join('');

    const currentPay = DATA.payments.find(p => p.id === basket.payId);
    let finalTotal = Math.round(subtotal * (1 - currentPay.disc)) + currentShipFee;
    document.getElementById('total-display').innerText = (subtotal > 0 ? finalTotal : 0) + " TL";
    
    const waBtn = document.getElementById('wa-btn');
    if(subtotal >= CONFIG.minLimit) {
        waBtn.classList.remove('disabled'); waBtn.innerText = "Siparişi Onayla";
    } else {
        waBtn.classList.add('disabled'); waBtn.innerText = subtotal === 0 ? "Ürün Ekleyin" : "Limit Bekleniyor";
    }
}

function orderWhatsApp(e) {
    e.preventDefault();
    const all = [...DATA.main, ...DATA.categories.flatMap(c => c.items)];
    let text = `*JetGo Elite Sipariş*\n\n`;
    let subtotal = 0;
    all.forEach(p => { if(basket.items[p.id]) { text += `• ${p.name} (x${basket.items[p.id]})\n`; subtotal += p.price * basket.items[p.id]; } });
    
    const currentShipFee = subtotal < CONFIG.shipLimit ? CONFIG.shipFee : 0;
    const pay = DATA.payments.find(p => p.id === basket.payId);
    let finalTotal = Math.round(subtotal * (1 - pay.disc)) + currentShipFee;

    text += `\n*Ödeme:* ${pay.name}\n*Teslimat:* ${currentShipFee === 0 ? 'Ücretsiz' : currentShipFee + ' TL'}\n*Genel Toplam:* ${finalTotal} TL`;
    text += `\n\nAdınızı, Soyadınızı, Adres ve Konum bilgilerinizi lütfen buradan gönderiniz.`;
    
    if(pay.id === 'eft') text += CONFIG.bankInfo;

    window.open(`https://wa.me/${CONFIG.phone.replace('+', '')}?text=${encodeURIComponent(text)}`);
}
